cmake_minimum_required(VERSION 3.16)
project(multiscale_hashgrid_demo LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(SDL2 REQUIRED)
find_package(OpenGL REQUIRED)
find_package(OpenMP)
add_executable(hashgrid_demo src/main.cpp)
target_link_libraries(hashgrid_demo PRIVATE SDL2::SDL2 OpenGL::GL)
if(OpenMP_CXX_FOUND)
  target_link_libraries(hashgrid_demo PRIVATE OpenMP::OpenMP_CXX)
endif()
